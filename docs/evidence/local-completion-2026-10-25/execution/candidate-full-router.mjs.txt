export async function verifyCandidate(){
 const manifest=JSON.parse(await readFile(manifestPath,'utf8'));
 const verify=async(base,files)=>{for(const [relative,expected]of Object.entries(files)){
  const path=resolve(base,relative);assert.ok(path.startsWith(base+sep),'snapshot_path_outside_boundary');
  assert.equal((await lstat(path)).isSymbolicLink(),false,'snapshot_link_changed');
  assert.equal(createHash('sha256').update(await readFile(path)).digest('hex'),expected,'snapshot_source_changed');
 }};
 await verify(candidatePath,manifest.candidateFiles);
 // Seed/expected domain logic imports the active checkout. It must still match
 // the source snapshot; frontend output is independently frozen in candidate/dist.
 await verify(resolve('.'),Object.fromEntries(Object.entries(manifest.sourceFilesBeforePatch).filter(([path])=>!path.startsWith('dist/'))));
 return manifest;
}
